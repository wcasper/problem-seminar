\begin{frame}{Recursive definition}
\begin{center}
\includegraphics[height=2in]{fig/xkcd-recursion}
\end{center}
Recursive definitions involves two things:
\begin{itemize}
\item a base case -- where the thing is defined explicitly
\item a recursive step -- a rule which relates the next case to previous cases
\end{itemize}
\end{frame}

\begin{frame}{Linear with constant coefficients}
Linearize:
$$\left[\begin{array}{c}u_{n+1}\\u_{n+2}\end{array}\right] = \left[\begin{array}{cc}
0 & 1\\
b & a
\end{array}\right] \left[\begin{array}{c}u_n\\u_{n+1}\end{array}\right]$$
\pause
If we let
$$\quad A = \left[\begin{array}{c}u_{n+1}\\u_{n+2}\end{array}\right] = \left[\begin{array}{cc}
0 & 1\\
b & a
\end{array}\right],\quad \vec u_n = \left[\begin{array}{c}u_n\\u_{n+1}\end{array}\right]$$
\pause
$$\vec u_{n+1} = A\vec u_n$$
\pause
$$\vec u_{n} = A\vec u_{n-1} = A(A\vec u_{n-2}) = A(A(A\vec u_{n-2})) = \dots$$
\pause
$$\vec u_n = A^n\vec u_0$$
\end{frame}

\begin{frame}{Fast Powers}
Polynomial long division:\\
{\small
$$\frac{x^{10}}{x^2-x-1} = 
x^8 + x^7 + 2 x^6 + 3 x^5 + 5 x^4 + 8 x^3 + 13 x^2 +  21 x + 34 + \frac{55 x + 34}{x^2 - x - 1}$$
}
\pause
$$x^{10} = (x^8 + x^7 + \dots + 21x + 34)(x^2-x-1) + 55x + 34$$
\pause
$$A^{10} = (A^8 + A^7 + \dots + 21A + 34I)(A^2-A^1-I) + 55A + 34I$$
\pause
$$A^{10} = \left[\begin{array}{cc}
0 & 0\\
0 & 0
\end{array}\right] + 55\left[\begin{array}{cc}
0 & 1\\
1 & 1
\end{array}\right] + 34\left[\begin{array}{cc}
1 & 0\\
0 & 1
\end{array}\right] = \left[\begin{array}{cc}
34 & 55\\
55 & 89
\end{array}\right]$$
\end{frame}

\begin{frame}{Faster Powers}
Only the polynomial remainder mattered!
\pause
\begin{block}{Question:}
Can we get the remainder \textbf{faster}?
\end{block}
\pause
$$x^{10} = (x^2-x-1)^2\cdot\text{poly} + cx + d$$
\pause
Roots of $x^2-x-1$ are $\varphi_+ = \frac{1+\sqrt{5}}{2}$ and $\varphi_- =  \frac{1-\sqrt{5}}{2}$
\pause
$$\varphi_+^{10} =  c\varphi_+ + d$$
$$\varphi_-^{10} =  c\varphi_- + d$$
\pause
$$c = \frac{\varphi_+^{10}-\varphi_-^{10}}{\varphi_+-\varphi_-} = 55,\quad d = \frac{\varphi_+^{10}-\varphi_-^{10}}{\varphi_+-\varphi_-} = 34$$
\end{frame}

\begin{frame}{Fibonacci closed form}
\pause
$$x^n = (x^2-x-1)^2\cdot\text{poly} + cx + d$$
\pause
$$\varphi_+^{n} =  c\varphi_+ + d$$
$$\varphi_-^{n} =  c\varphi_- + d$$
\pause
$$c = \frac{\varphi_+^{n}-\varphi_-^{n}}{\varphi_+-\varphi_-},\quad d = \frac{\varphi_+^{n-1}-\varphi_-^{n-1}}{\varphi_+-\varphi_-}$$
\pause
$$A^n = \frac{\varphi_+^{n}-\varphi_-^{n}}{\varphi_+-\varphi_-}A + \frac{\varphi_+^{n-1}-\varphi_-^{n-1}}{\varphi_+-\varphi_-}I$$
\pause
$$F_n = \frac{\varphi_+^n - \varphi_-^n}{\varphi_+-\varphi_-}.$$
\end{frame}

\begin{frame}{Generalization}

Determine the limit of $u_{n}$.
\end{frame}

\begin{frame}{Putnam 1993 A2}
Let $z_0$ and $z_1$ be real numbers.

$$z_{n}^2-z_{n+1}z_{n-1} = 1,\ \ n \geq 1.$$
Show that there is a constant $a\in \mathbb R$ with
$$z_{n+1} + z_{n-1} = az_n,\ \ n\geq 1$$
Can you find an explicit expression for $z_n$?
\end{frame}
